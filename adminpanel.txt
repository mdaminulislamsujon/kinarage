<script>

document.addEventListener(
    "DOMContentLoaded",
    () => {


        // ==========================================
        // CONFIG
        // ==========================================

        const API_URL =
            "http://localhost:5000/api";


        const SERVER_URL =
            "http://localhost:5000";



        // ==========================================
        // ELEMENTS
        // ==========================================

        const adminUser =
            document.getElementById(
                "adminUser"
            );


        const logoutButton =
            document.getElementById(
                "logoutButton"
            );


        const homeButton =
            document.getElementById(
                "homeButton"
            );


        const productForm =
            document.getElementById(
                "productForm"
            );


        const formTitle =
            document.getElementById(
                "formTitle"
            );


        const submitButton =
            document.getElementById(
                "submitButton"
            );


        const cancelEditButton =
            document.getElementById(
                "cancelEditButton"
            );


        const formMessage =
            document.getElementById(
                "formMessage"
            );


        const productImage =
            document.getElementById(
                "productImage"
            );


        const productImageURL =
            document.getElementById(
                "productImageURL"
            );


        const imagePreview =
            document.getElementById(
                "imagePreview"
            );


        const productsLoading =
            document.getElementById(
                "productsLoading"
            );


        const productsTableWrapper =
            document.getElementById(
                "productsTableWrapper"
            );


        const productsTableBody =
            document.getElementById(
                "productsTableBody"
            );



        // ==========================================
        // STATE
        // ==========================================

        let editingProductId = null;

        let productsData = [];

        let currentPreviewURL = null;



        // ==========================================
        // GET TOKEN
        // ==========================================

        function getToken() {

            return localStorage.getItem(
                "kinarage-token"
            );

        }



        // ==========================================
        // GET SAVED USER
        // ==========================================

        function getSavedUser() {

            try {

                const value =
                    localStorage.getItem(
                        "kinarage-user"
                    );


                return value
                    ? JSON.parse(value)
                    : null;

            } catch {

                return null;

            }

        }



        // ==========================================
        // ESCAPE HTML
        // ==========================================

        function escapeHTML(value) {

            return String(value ?? "")
                .replace(
                    /&/g,
                    "&amp;"
                )
                .replace(
                    /</g,
                    "&lt;"
                )
                .replace(
                    />/g,
                    "&gt;"
                )
                .replace(
                    /"/g,
                    "&quot;"
                )
                .replace(
                    /'/g,
                    "&#039;"
                );

        }



        // ==========================================
        // IMAGE URL
        // ==========================================

        function getImageURL(
            imageURL
        ) {

            if (!imageURL) {

                return null;

            }


            if (
                imageURL.startsWith(
                    "http://"
                ) ||
                imageURL.startsWith(
                    "https://"
                )
            ) {

                return imageURL;

            }


            return (
                SERVER_URL +
                imageURL
            );

        }



        // ==========================================
        // SHOW MESSAGE
        // ==========================================

        function showMessage(
            message,
            type = "success"
        ) {

            formMessage.textContent =
                message;


            formMessage.className =
                `message ${type}`;

        }



        // ==========================================
        // AUTH CHECK
        // ==========================================

        async function checkAdmin() {

            const token =
                getToken();


            const savedUser =
                getSavedUser();


            if (
                !token ||
                !savedUser ||
                savedUser.role !== "admin"
            ) {

                window.location.href =
                    "admin-login.html";

                return false;

            }



            try {

                const response =
                    await fetch(
                        `${API_URL}/auth/me`,
                        {
                            headers: {
                                Authorization:
                                    `Bearer ${token}`
                            }
                        }
                    );


                if (!response.ok) {

                    throw new Error(
                        "Authentication failed"
                    );

                }


                const data =
                    await response.json();


                if (
                    !data.user ||
                    data.user.role !== "admin"
                ) {

                    throw new Error(
                        "Admin access required"
                    );

                }


                adminUser.innerHTML =
                    `👑 ${escapeHTML(
                        data.user.name
                    )} · ${escapeHTML(
                        data.user.email
                    )}`;


                return true;


            } catch (error) {

                console.error(
                    "Admin authentication error:",
                    error
                );


                localStorage.removeItem(
                    "kinarage-token"
                );


                localStorage.removeItem(
                    "kinarage-user"
                );


                window.location.href =
                    "admin-login.html";


                return false;

            }

        }



        // ==========================================
        // LOGOUT
        // ==========================================

        logoutButton.addEventListener(
            "click",
            () => {

                localStorage.removeItem(
                    "kinarage-token"
                );


                localStorage.removeItem(
                    "kinarage-user"
                );


                window.location.href =
                    "admin-login.html";

            }
        );



        // ==========================================
        // HOME
        // ==========================================

        homeButton.addEventListener(
            "click",
            () => {

                window.location.href =
                    "index.html";

            }
        );



        // ==========================================
        // LAPTOP IMAGE PREVIEW
        // ==========================================

        productImage.addEventListener(
            "change",
            () => {

                const file =
                    productImage.files[0];


                if (!file) {

                    return;

                }



                // Maximum 5MB

                if (
                    file.size >
                    5 * 1024 * 1024
                ) {

                    productImage.value = "";


                    imagePreview.innerHTML = `
                        <div class="no-image">
                            ❌ Image must be less than 5MB
                        </div>
                    `;


                    return;

                }



                // If laptop image selected,
                // remove URL input

                productImageURL.value = "";



                // Revoke old preview

                if (currentPreviewURL) {

                    URL.revokeObjectURL(
                        currentPreviewURL
                    );

                }



                currentPreviewURL =
                    URL.createObjectURL(
                        file
                    );


                imagePreview.innerHTML = `
                    <img
                        src="${currentPreviewURL}"
                        alt="Product preview"
                    >
                `;

            }
        );



        // ==========================================
        // IMAGE URL PREVIEW
        // ==========================================

        productImageURL.addEventListener(
            "input",
            () => {

                const url =
                    productImageURL.value.trim();


                if (!url) {

                    imagePreview.innerHTML = `
                        <div class="no-image">
                            📷<br>
                            No image selected
                        </div>
                    `;

                    return;

                }



                // URL ব্যবহার করলে
                // laptop file remove

                productImage.value = "";



                imagePreview.innerHTML = `
                    <img
                        src="${escapeHTML(url)}"
                        alt="Product preview"
                        onerror="
                            this.style.display='none';
                            this.nextElementSibling.style.display='block';
                        "
                    >

                    <div
                        class="no-image"
                        style="display:none;"
                    >
                        ❌ Image URL could not be loaded
                    </div>
                `;

            }
        );



        // ==========================================
        // LOAD PRODUCTS
        // ==========================================

        async function loadProducts() {

            productsLoading.style.display =
                "block";


            productsTableWrapper.style.display =
                "none";


            try {

                const response =
                    await fetch(
                        `${API_URL}/products`
                    );


                if (!response.ok) {

                    throw new Error(
                        `HTTP ${response.status}`
                    );

                }


                productsData =
                    await response.json();


                renderProducts();


            } catch (error) {

                console.error(
                    "Load products error:",
                    error
                );


                productsLoading.innerHTML = `
                    ❌ Failed to load products.
                    <br><br>
                    Make sure Node.js server is running.
                `;

            }

        }



        // ==========================================
        // RENDER PRODUCTS
        // ==========================================

        function renderProducts() {

            productsLoading.style.display =
                "none";


            productsTableWrapper.style.display =
                "block";


            if (
                !productsData ||
                productsData.length === 0
            ) {

                productsTableBody.innerHTML = `
                    <tr>
                        <td
                            colspan="7"
                            class="empty-state"
                        >
                            No products found.
                        </td>
                    </tr>
                `;


                return;

            }



            productsTableBody.innerHTML =
                productsData
                    .map(
                        (product) => {

                            const imageURL =
                                getImageURL(
                                    product.image_url
                                );


                            const imageHTML =
                                imageURL
                                    ? `
                                        <img
                                            class="product-thumb"
                                            src="${escapeHTML(
                                                imageURL
                                            )}"
                                            alt="${escapeHTML(
                                                product.name
                                            )}"
                                            onerror="
                                                this.style.display='none';
                                                this.nextElementSibling.style.display='flex';
                                            "
                                        >

                                        <div
                                            class="no-thumb"
                                            style="display:none;"
                                        >
                                            📦
                                        </div>
                                    `
                                    : `
                                        <div class="no-thumb">
                                            📦
                                        </div>
                                    `;


                            return `
                                <tr>

                                    <td>
                                        ${imageHTML}
                                    </td>


                                    <td>
                                        <strong>
                                            ${escapeHTML(
                                                product.name
                                            )}
                                        </strong>
                                    </td>


                                    <td>
                                        ${escapeHTML(
                                            product.category
                                        )}
                                    </td>


                                    <td>
                                        <strong>
                                            ${escapeHTML(
                                                product.price
                                            )}
                                        </strong>
                                    </td>


                                    <td>
                                        ${escapeHTML(
                                            product.rating ||
                                            "N/A"
                                        )}
                                    </td>


                                    <td>
                                        ${escapeHTML(
                                            product.processor ||
                                            "N/A"
                                        )}
                                    </td>


                                    <td>

                                        <div
                                            class="action-buttons"
                                        >

                                            <button
                                                class="edit-btn"
                                                data-id="${product.id}"
                                            >
                                                ✏️ Edit
                                            </button>


                                            <button
                                                class="delete-btn"
                                                data-id="${product.id}"
                                            >
                                                🗑️ Delete
                                            </button>

                                        </div>

                                    </td>

                                </tr>
                            `;

                        }
                    )
                    .join("");



            // ======================================
            // EDIT BUTTONS
            // ======================================

            document
                .querySelectorAll(
                    ".edit-btn"
                )
                .forEach(
                    (button) => {

                        button.addEventListener(
                            "click",
                            () => {

                                const id =
                                    button.dataset.id;


                                startEdit(
                                    Number(id)
                                );

                            }
                        );

                    }
                );



            // ======================================
            // DELETE BUTTONS
            // ======================================

            document
                .querySelectorAll(
                    ".delete-btn"
                )
                .forEach(
                    (button) => {

                        button.addEventListener(
                            "click",
                            () => {

                                const id =
                                    button.dataset.id;


                                deleteProduct(
                                    Number(id)
                                );

                            }
                        );

                    }
                );

        }



        // ==========================================
        // START EDIT
        // ==========================================

        function startEdit(id) {

            const product =
                productsData.find(
                    (item) =>
                        Number(item.id) ===
                        Number(id)
                );


            if (!product) {

                alert(
                    "Product not found."
                );

                return;

            }


            editingProductId =
                product.id;


            formTitle.textContent =
                `✏️ Edit Product #${product.id}`;


            submitButton.textContent =

                    window.location.href =

                    return;

                }

                // FORM DATA

                    document

                        .trim()

                    document
                        .getElementById(

                const url =

                            "Admin access required.",

                        );

                        );

                        isEditing

                    );

                        error
                    );

                        error.message ||
                        "Something went wrong.",
                        "error"
                    );

                        false;


                    cancelEditButton.disabled =
                        false;

                        editingProductId !== null
                            ? "💾 Update Product"
                            : "➕ Add Product";

        async function deleteProduct(id) {

            const product =
                productsData.find(
                    (item) =>
                        Number(item.id) ===
                        Number(id)
                );


            if (!product) {

                return;

            }



            const confirmed =
                confirm(
                    `Are you sure you want to delete "${product.name}"?`
                );


            if (!confirmed) {

                return;

            }



            const token =
                getToken();


            if (!token) {

                window.location.href =
                    "admin-login.html";

                return;

            }



            try {

                const response =
                    await fetch(
                        `${API_URL}/products/${id}`,
                        {
                            method: "DELETE",

                            headers: {
                                Authorization:
                                    `Bearer ${token}`
                            }
                        }
                    );



                const data =
                    await response.json();



                // ==================================
                // TOKEN EXPIRED
                // ==================================

                if (
                    response.status ===
                    401
                ) {

                    localStorage.removeItem(
                        "kinarage-token"
                    );


                    localStorage.removeItem(
                        "kinarage-user"
                    );


                    window.location.href =
                        "admin-login.html";


                    return;

                }



                if (!response.ok) {

                    throw new Error(
                        data.error ||
                        "Failed to delete product"
                    );

                }



                // ==================================
                // RESET EDIT MODE
                // ==================================

                if (
                    Number(editingProductId) ===
                    Number(id)
                ) {

                    resetForm();

                }



                await loadProducts();



            } catch (error) {

                console.error(
                    "Delete product error:",
                    error
                );


                alert(
                    error.message ||
                    "Failed to delete product."
                );

            }

        }
